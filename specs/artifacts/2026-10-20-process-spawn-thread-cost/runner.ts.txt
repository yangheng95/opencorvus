export async function run(items:any[],executable:string,concurrency:number){
 const rows:Row[]=[],started=performance.now()
 const outcomes=await settledWork({items,concurrency,run:async(item:any,index:number)=>{
  const row:Row={index,spawnMs:0,closes:[]};rows[index]=row
  return context.run(row,async()=>{
   const at=performance.now()
   const result=await Process.runHost([executable,'-c','core.longPaths=true','rev-parse','--path-format=absolute','--show-toplevel','--git-common-dir'],{cwd:item.worktree,env:{GIT_CEILING_DIRECTORIES:path.dirname(item.worktree)},stdin:'ignore',owner:'process-spawn-thread-cost-475',inactivityTimeoutMs:15000,onSpawned:()=>{row.readyMs=performance.now()-at}})
   row.ms=performance.now()-at;row.code=result.code;row.stdout=result.stdout.toString();row.stderr=result.stderr.toString()
   assert.equal(result.code,0)
   const paths=row.stdout.replace(/\r?\n$/,'').split(/\r?\n/);assert.equal(paths.length,2)
   assert.equal(await fs.realpath(paths[0]!),await fs.realpath(item.worktree))
   assert((await fs.stat(paths[1]!)).isDirectory())
   assert.deepEqual(row.closes,[{code:0,signal:null}]);assert(row.request)
   const remaining=await fs.readdir(path.dirname(row.request.dir));assert.equal(remaining.includes(path.basename(row.request.dir)),false)
   return{index,stdout:row.stdout,stderr:row.stderr}
  })
 }})

 return{ms:performance.now()-started,syncSpawnMs:rows.reduce((n,row)=>n+row.spawnMs,0),rows,live,failures,pid:process.pid}
}
