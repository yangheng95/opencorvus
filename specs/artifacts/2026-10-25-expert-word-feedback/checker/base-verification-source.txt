  for(;;){const response=await request(`/task/${current.task.id}/artifact-read`,{locator,byte_offset:offset,max_bytes:4096});const range=response.headers.get("content-range")!;ranges.push(range);const chunk=Buffer.from(await response.arrayBuffer());chunks.push(chunk);const total=Number(range.split("/")[1]);offset+=chunk.length;if(offset===total)break;assert(chunk.length>0&&offset<total)}

}
