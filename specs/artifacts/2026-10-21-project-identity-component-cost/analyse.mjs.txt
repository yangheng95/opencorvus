  const phases=row.result.native_phases??[]

 }
 return{run,wallMs:data.wallMs,blockingRenameWindows:stats(profile.samples.map(row=>row.finishedAt-row.startedAt)),clientOverlappingWaits:Object.fromEntries(Object.entries(stages).map(([name,a])=>[name,stats(a)])),nativeOverlappingPhases:Object.fromEntries(Object.entries(nativePhases).map(([name,a])=>[name,stats(a)])),boundary:'Only synchronous native clock samples are serialized main-thread intervals.
