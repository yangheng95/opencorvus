param(
  [Parameter(Mandatory=$true)][ValidateSet('failure','preparation-deadline','manual-join')][string]$Case,
  [Parameter(Mandatory=$true)][string]$RunRoot,
  [Parameter(Mandatory=$true)][ValidateRange(1,65535)][int]$Port,
  [Parameter(Mandatory=$true)][string]$EvidenceRoot
)
$ErrorActionPreference='Stop'
$taskRun=[IO.Path]::GetFullPath($RunRoot)
$taskFormalEvidence=[IO.Path]::GetFullPath($EvidenceRoot)
$taskQualificationPort=$Port
. (Join-Path $PSScriptRoot 'live-sol-launch.ps1') -FunctionsOnly
$taskPrefix='owned-settlement-19-'+$Case
$taskOccurrence=$taskPrefix+'-'+[Guid]::NewGuid().ToString()
$taskBun='C:/Users/hengu/.bun/bin/bun.exe'
$taskPackage='D:/myhexin-local/opencorvus/packages/opencorvus'
if(Test-Path -LiteralPath $taskRun){throw 'Fresh no-Provider qualification directory required'}
if(@(Get-NetTCPConnection -ErrorAction Stop|Where-Object {$_.State -eq 'Listen' -and $_.LocalPort -eq $taskQualificationPort}).Count){throw 'Qualification port occupied'}
$taskRuntime=Join-Path $taskRun 'runtime'
$taskProject=Join-Path $taskRun 'project'
$taskLocalEvidence=Join-Path $taskRun 'evidence'
foreach($directory in @($taskRuntime,$taskProject,$taskLocalEvidence,$taskFormalEvidence,(Join-Path $taskRuntime 'config'),(Join-Path $taskRun 'home'),(Join-Path $taskRun 'appdata'),(Join-Path $taskRun 'localappdata'),(Join-Path $taskRun 'temp'))){[IO.Directory]::CreateDirectory($directory)|Out-Null}
[IO.File]::WriteAllText((Join-Path $taskRuntime 'config/opencorvus.jsonc'),'{"prompt_profile":{"active":"base"}}',[Text.UTF8Encoding]::new($false))
$taskEnvironmentBefore=@{}
Get-ChildItem Env:|ForEach-Object {$taskEnvironmentBefore[$_.Name]=$_.Value}
$taskEnvironment=@{}
foreach($name in @('PATH','SystemRoot','WINDIR','ComSpec','PATHEXT','LANG','LC_ALL','LC_CTYPE')){if($taskEnvironmentBefore.ContainsKey($name)){$taskEnvironment[$name]=$taskEnvironmentBefore[$name]}}
$taskEnvironment.HOME=Join-Path $taskRun 'home'
$taskEnvironment.USERPROFILE=$taskEnvironment.HOME
$taskEnvironment.HOMEDRIVE='C:'
$taskEnvironment.HOMEPATH=$taskEnvironment.HOME.Substring(2).Replace('/','\')
$taskEnvironment.APPDATA=Join-Path $taskRun 'appdata'
$taskEnvironment.LOCALAPPDATA=Join-Path $taskRun 'localappdata'
$taskEnvironment.TEMP=Join-Path $taskRun 'temp'
$taskEnvironment.TMP=$taskEnvironment.TEMP
$taskEnvironment.TMPDIR=$taskEnvironment.TEMP
$taskEnvironment.OPENCORVUS_HOME=$taskRuntime
$taskEnvironment.OPENCORVUS_DISABLE_EXTERNAL_SKILLS='1'
$taskEnvironment.OPENCORVUS_DISABLE_AUTOUPDATE='1'
$taskEnvironment.OPENCORVUS_TASK_PROCESS_MODE='native'
foreach($name in @('CONFIG','DATA','CACHE','STATE')){$taskEnvironment['XDG_'+$name+'_HOME']=Join-Path $taskRun ('xdg-'+$name.ToLower());[IO.Directory]::CreateDirectory($taskEnvironment['XDG_'+$name+'_HOME'])|Out-Null}
$taskProcess=$null
$taskReceipt=$null
$taskJobs=@()
try {
  try {
    Get-ChildItem Env:|ForEach-Object {[Environment]::SetEnvironmentVariable($_.Name,$null,'Process')}
    foreach($name in $taskEnvironment.Keys){[Environment]::SetEnvironmentVariable($name,$taskEnvironment[$name],'Process')}
    Push-Location $taskPackage
    try {
      $taskNativeText=& $taskBun -e 'import {observedProcessOccurrence} from "./src/runtime/process-occurrence.ts"; const parent=observedProcessOccurrence(process.ppid); if(!parent) throw new Error("Exact qualification parent required"); process.stdout.write(JSON.stringify(parent));'
      if($LASTEXITCODE -ne 0){throw 'Native parent occurrence qualification failed'}
      $taskNativeParent=$taskNativeText|ConvertFrom-Json
    } finally {Pop-Location}
    if($taskNativeParent.pid -ne $PID){throw 'Observed native parent is not this checker'}
    $taskStartup=Join-Path $taskRun 'startup.json'
    $taskArgs=@('src/index.ts','serve','--hostname','127.0.0.1','--port',[string]$taskQualificationPort,'--project-dir',$taskProject,'--startup-receipt',$taskStartup,'--startup-occurrence',$taskOccurrence,'--parent-pid',[string]$taskNativeParent.pid,'--parent-process-instance-id',$taskNativeParent.processInstanceID,'--log-level','DEBUG')
    if(@($taskArgs|Where-Object {$_ -match '[\s"]'}).Count){throw 'Qualification launch requires quoting review'}
    $taskProcess=Start-Process -FilePath $taskBun -ArgumentList $taskArgs -WorkingDirectory $taskPackage -WindowStyle Hidden -RedirectStandardOutput (Join-Path $taskRun 'stdout.log') -RedirectStandardError (Join-Path $taskRun 'stderr.log') -PassThru
  } finally {
    Get-ChildItem Env:|ForEach-Object {[Environment]::SetEnvironmentVariable($_.Name,$null,'Process')}
    foreach($name in $taskEnvironmentBefore.Keys){[Environment]::SetEnvironmentVariable($name,$taskEnvironmentBefore[$name],'Process')}
  }
  $taskOwner=Get-CimInstance Win32_Process -Filter "ProcessId=$($taskProcess.Id)"|Select-Object ProcessId,ParentProcessId,CreationDate,ExecutablePath,CommandLine
  if(!$taskOwner.CommandLine -or $taskOwner.ParentProcessId -ne $PID){throw 'Exact direct qualification child required'}
  $taskChildren=@(Get-CimInstance Win32_Process -Filter "ParentProcessId=$($taskProcess.Id)"|Where-Object {([DateTime]$_.CreationDate).ToUniversalTime() -ge ([DateTime]$taskOwner.CreationDate).ToUniversalTime()}|Select-Object ProcessId,ParentProcessId,CreationDate,ExecutablePath,CommandLine)
  $taskReceipt=[ordered]@{observedAtUtc=[DateTime]::UtcNow.ToString('o');runRoot=$taskRun;runtime=$taskRuntime;project=$taskProject;evidence=$taskLocalEvidence;settlementEvidenceRoot=$taskFormalEvidence;evidencePrefix=$taskPrefix;occurrence=$taskOccurrence;port=$taskQualificationPort;owner=$taskOwner;observedProcessChain=@($taskOwner)+$taskChildren;parent=$taskNativeParent;qualification='Actual no-Provider development serve; no Task/Session seeded, no copied authority'}
  Write-OwnedLaunchFact (Join-Path $taskRun 'launch-owner.json') $taskReceipt
  $taskReceipt=Get-Content -Raw -LiteralPath (Join-Path $taskRun 'launch-owner.json')|ConvertFrom-Json
  $taskReadyDeadline=[DateTime]::UtcNow.AddSeconds(60)
  do {
    if(Test-Path -LiteralPath $taskStartup){
      $taskReady=Get-Content -Raw -LiteralPath $taskStartup|ConvertFrom-Json
      if($taskReady.outcome -ne 'listening' -or $taskReady.pid -ne $taskProcess.Id -or $taskReady.occurrenceID -ne $taskOccurrence){throw 'Genuine startup identity failed'}
      break
    }
    $taskProcess.Refresh()
    if($taskProcess.HasExited){throw 'Actual serve exited before readiness'}
    Start-Sleep -Milliseconds 250
  }while([DateTime]::UtcNow -lt $taskReadyDeadline)
  if(!$taskReady){throw 'Actual serve readiness timed out'}
  $taskHealth=Invoke-WebRequest -UseBasicParsing -Uri "http://127.0.0.1:$taskQualificationPort/global/health" -TimeoutSec 15
  $taskHealthBody=$taskHealth.Content|ConvertFrom-Json
  if($taskHealth.StatusCode -ne 200 -or !$taskHealthBody.healthy){throw 'Actual service health contract failed'}
  Write-OwnedLaunchFact (Join-Path $taskLocalEvidence 'qualification-health.json') @{status=[int]$taskHealth.StatusCode;body=$taskHealthBody;occurrence=$taskOccurrence;observedAtUtc=[DateTime]::UtcNow.ToString('o')}
  $taskBorn=([DateTimeOffset]$taskReceipt.owner.CreationDate).ToUnixTimeMilliseconds()
  $taskPreparationDeadline=$taskBorn+600000
  if($Case -eq 'failure'){
    # Explicit checker fault injection; not a product or Provider failure.
    Write-OwnedLaunchFact (Join-Path $taskLocalEvidence 'failed.json') @{pid=$taskOwner.ProcessId;occurrence=$taskOccurrence;error=@{name='OwnedCheckerQualificationFailure';message='Explicit first-failure settlement qualification after actual healthy200'};qualification='Local supervisor control input; no model/Task pass or product defect claim'}
  }elseif($Case -eq 'preparation-deadline'){
    $taskPreparationDeadline=[DateTimeOffset]::UtcNow.ToUnixTimeMilliseconds()+10000
    Write-OwnedLaunchFact (Join-Path $taskLocalEvidence 'qualification-deadline-policy.json') @{kind='short-no-Provider-preparation-policy';preparationDeadlineMs=$taskPreparationDeadline;genuineTaskMaximumMs=900000;taskBoundary='No Task is seeded; actual Task total remains unqualified by this case'}
  }else{
    $taskHelper=Join-Path $PSScriptRoot 'live-sol-shutdown.ps1'
    foreach($index in 1..2){
      $taskJobs+=Start-Job -ScriptBlock {param($run,$evidence,$prefix,$helper,$reason);& $helper -RunRoot $run -EvidenceRoot $evidence -EvidencePrefix $prefix -Reason $reason} -ArgumentList $taskRun,$taskFormalEvidence,$taskPrefix,$taskHelper,('qualified-manual-'+$index)
    }
  }
  Watch-OwnedQualification -LaunchReceipt $taskReceipt -EvidenceRoot $taskFormalEvidence -PreparationDeadlineMilliseconds $taskPreparationDeadline|Out-Null
  foreach($job in $taskJobs){
    $finished=Wait-Job -Job $job -Timeout 60
    if(!$finished -or $job.State -ne 'Completed'){throw 'Concurrent manual settlement did not complete'}
    $joinedOutput=@(Receive-Job -Job $job -ErrorAction Stop)
    Write-OwnedLaunchFact (Join-Path $taskLocalEvidence ('manual-job-'+$job.Id+'.json')) @{state=$job.State.ToString();output=$joinedOutput;qualification='Actual competing callers join the same physical receipt'}
    Remove-Job -Job $job
  }
  $taskPhysical=Get-Content -Raw -LiteralPath (Join-Path $taskFormalEvidence ($taskPrefix+'-physical-terminal-and-pair-cleanup.json'))|ConvertFrom-Json
  $taskAdmission=Get-Content -Raw -LiteralPath (Join-Path $taskFormalEvidence ($taskPrefix+'-shutdown-admission.json'))|ConvertFrom-Json
  if(!$taskPhysical.physicalCompletion -or !$taskPhysical.pairedCleanupComplete -or @($taskPhysical.chain|Where-Object originalOccurrencePresent).Count -or @($taskPhysical.listeners|Where-Object {$_}).Count){throw 'Real chain/port/pair contract failed'}
  if(!$taskAdmission.dispatch -or $taskAdmission.status -ne 200 -or $taskAdmission.pageClosed){throw 'Actual public shutdown admission failed'}
  foreach($name in @('auth.json','models.json')){if(Test-Path -LiteralPath (Join-Path $taskRuntime ('data/'+$name))){throw 'Unexpected no-Provider authority member'}}
  $taskProcess.Refresh()
  if(!$taskProcess.HasExited){throw 'Original actual child still live after closure'}
  $taskTrigger=$null
  if(Test-Path -LiteralPath (Join-Path $taskLocalEvidence 'supervision-settlement-trigger.json')){$taskTrigger=Get-Content -Raw -LiteralPath (Join-Path $taskLocalEvidence 'supervision-settlement-trigger.json')|ConvertFrom-Json}
  if($Case -eq 'failure' -and $taskTrigger.reason -ne 'immutable-wrapper-failure'){throw 'First-failure trigger contract mismatch'}
  if($Case -eq 'preparation-deadline' -and ($taskTrigger.reason -ne 'fixed-preparation-maximum' -or $taskTrigger.preparationDeadlineMs -ne $taskPreparationDeadline)){throw 'Fixed preparation trigger contract mismatch'}
  & (Join-Path $PSScriptRoot 'live-sol-shutdown.ps1') -RunRoot $taskRun -EvidenceRoot $taskFormalEvidence -EvidencePrefix $taskPrefix -Reason 'post-completion-join'|Out-Null
  Write-OwnedLaunchFact (Join-Path $taskFormalEvidence ($taskPrefix+'-qualified.json')) @{case=$Case;occurrence=$taskOccurrence;observedAtUtc=[DateTime]::UtcNow.ToString('o');healthStatus=200;shutdownStatus=$taskAdmission.status;exitCode=$taskProcess.ExitCode;physicalCompletion=$taskPhysical.physicalCompletion;pairedCleanupComplete=$taskPhysical.pairedCleanupComplete;trigger=$taskTrigger;boundary='Real existing serve and sole supervisor/shutdown; no Provider, seeded Task, UI automation, fake HTTP server, timer extension or process kill'}
  Write-Output "Qualified actual no-Provider settlement case $Case on port $taskQualificationPort."
} catch {
  $taskOriginalFailure=$_
  if($taskReceipt){
    try {& (Join-Path $PSScriptRoot 'live-sol-shutdown.ps1') -RunRoot $taskRun -EvidenceRoot $taskFormalEvidence -EvidencePrefix $taskPrefix -Reason 'qualification-checker-error'|Out-Null}catch {Write-Output 'Exact qualification cleanup remains blocked; retained original evidence and authority boundary.'}
  }
  throw $taskOriginalFailure
}
