import { Database } from "bun:sqlite"
import fs from "node:fs"
import path from "node:path"
const [database, evidenceRoot, prefix, phase] = process.argv.slice(2)
if (!database || !evidenceRoot || !prefix || !path.isAbsolute(database) || !path.isAbsolute(evidenceRoot) || !/^[a-z0-9-]+$/.test(prefix)) throw new Error("Explicit absolute owned database/evidence and prefix required")
if (!phase || !/^[a-z0-9-]+$/.test(phase)) throw new Error("Explicit evidence phase required")
const db = new Database(database, { readonly: true })
const facts = db.transaction(() => {
  const rows = (sql: string) => db.query(sql).all()
  const owners = rows("SELECT o.session_id,o.project_id,o.directory,o.generation,o.owner_pid,o.owner_process_instance_id,o.owner_occurrence_id,o.time_acquired,s.kind,s.parent_id,s.project_id AS actual_session_project FROM session_prompt_owner o LEFT JOIN session s ON s.id=o.session_id ORDER BY o.session_id")
  const messages = rows("SELECT id,session_id,time_created,json_extract(data,'$.role') AS role,json_extract(data,'$.parentID') AS parent_id,json_extract(data,'$.acceptedInputMessageIDs') AS accepted_input_ids,json_extract(data,'$.time.created') AS created_at,json_extract(data,'$.time.completed') AS completed_at,json_extract(data,'$.finish') AS finish,json_extract(data,'$.error.name') AS error_name FROM message ORDER BY time_created,id")
  const events = rows("SELECT id,kind,type,aggregate_type,aggregate_id,project_id,task_id,session_id,causation_id,correlation_id,reply_to,seq,deadline_ms,emitted_at,json_extract(payload,'$.execution_epoch') AS execution_epoch FROM protocol_event ORDER BY emitted_at,seq,id")
  const inbox = rows("SELECT id,envelope_id,actor,actor_id,visible_at,time_created FROM protocol_inbox ORDER BY time_created,id")
  const deliveryReceipts = rows("SELECT id,inbox_id,time_created,json_extract(receipt,'$.kind') AS kind,json_extract(receipt,'$.visible_at') AS retry_visible_at,json_extract(receipt,'$.message_id') AS message_id,json_extract(receipt,'$.ingress_id') AS ingress_id,json_extract(receipt,'$.closure_event_id') AS closure_event_id,json_extract(receipt,'$.error_name') AS error_name FROM protocol_delivery_receipt ORDER BY time_created,id")
  const unresolvedInbox = rows("SELECT i.id,i.envelope_id,i.actor,i.actor_id,i.visible_at FROM protocol_inbox i WHERE NOT EXISTS(SELECT 1 FROM protocol_delivery_receipt r WHERE r.inbox_id=i.id AND json_extract(r.receipt,'$.kind') <> 'retry_wait') ORDER BY i.visible_at,i.id")
  const leases = rows("SELECT id,target,target_id,owner_occurrence_id,time_activated,expires_at FROM engine_control_activation_lease ORDER BY time_activated,id")
  const leaseGrants = rows("SELECT lease_id,ordinal,expires_at,time_created FROM engine_control_activation_lease_grant ORDER BY lease_id,ordinal")
  const tasks = rows("SELECT id,project_id,session_id,request_id,source,product_pillar,time_archived,time_created FROM engine_task ORDER BY time_created,id")
  const taskIngress = rows("SELECT id,task_id,project_id,execution_epoch,sequence,source,source_id,policy_id,time_accepted FROM engine_task_root_ingress ORDER BY task_id,sequence,id")
  const taskIngressPolicies = rows("SELECT id,task_id,project_id,execution_epoch,sequence,source,source_id,policy_id,time_accepted FROM engine_task_root_ingress_policy ORDER BY task_id,sequence,id")
  const taskWaits = rows("SELECT id,task_id,project_id,execution_epoch,due_at,creator_ingress_id,creator_activation_id,time_created FROM engine_task_wait_registration ORDER BY time_created,id")
  const taskWaitSettlements = rows("SELECT wait_id,ingress_id,disposition,time_created FROM engine_task_wait_settlement ORDER BY time_created,wait_id")
  const sessionControls = rows("SELECT id,session_id,kind,source,time_created FROM session_control_record ORDER BY time_created,id")
  const sessionControlEvents = rows("SELECT id,control_id,kind,time_created FROM session_control_event ORDER BY time_created,id")
  const missionClosures = rows("SELECT id,type,aggregate_id AS session_id,project_id,correlation_id AS operation_id,seq,emitted_at FROM protocol_event WHERE type IN ('mission.execution.opened','mission.execution.closing','mission.execution.closed') ORDER BY aggregate_id,seq,id")
  const automationDefinitions = rows("SELECT id,definition_id,revision,project_id,session_id,kind,scope,execution_mode,status,due_at,time_created FROM automation ORDER BY definition_id,revision,id")
  const automationTombstones = rows("SELECT id,definition_id,revision,time_created FROM automation_definition_tombstone ORDER BY definition_id,revision,id")
  const automationTargets = rows("SELECT automation_revision_id,project_id,position FROM automation_project_target ORDER BY automation_revision_id,position")
  const automationFires = rows("SELECT id,automation_revision_id,scheduled_due_at,origin,time_created FROM automation_fire ORDER BY time_created,id")
  const automationFrontiers = rows("SELECT definition_id,automation_revision_id,fire_id,available_at FROM automation_fire_frontier ORDER BY definition_id")
  const automationAttempts = rows("SELECT id,fire_id,ordinal,owner_occurrence_id,lease_id,lease_grant_ordinal,lease_expires_at,time_created FROM automation_fire_attempt ORDER BY fire_id,ordinal")
  const automationAttemptReceipts = rows("SELECT attempt_id,outcome,retry_at,time_created FROM automation_fire_attempt_receipt ORDER BY time_created,attempt_id")
  const automationRuns = rows("SELECT id,automation_revision_id,fire_id,target_project_id,mission_opened_event_id,mission_disposition,mission_closure_event_id,started_at FROM automation_run ORDER BY started_at,id")
  const automationRunReceipts = rows("SELECT id,run_id,outcome,disposition,closure_event_id,retry_at,time_created FROM automation_run_receipt ORDER BY time_created,id")
  const automationDelaySettlements = rows("SELECT definition_id,disposition,assistant_message_id,accepted_input_message_ids,fire_id,time_created FROM automation_delay_settlement ORDER BY time_created,definition_id")
  const eventDefinitions = rows("SELECT id,definition_id,revision,project_id,session_id,event_type,enabled,one_shot,cooldown_ms,time_created FROM event_job ORDER BY definition_id,revision,id")
  const eventTombstones = rows("SELECT id,definition_id,revision,time_created FROM event_job_definition_tombstone ORDER BY definition_id,revision,id")
  const eventOccurrences = rows("SELECT id,bus_outbox_id,project_id,event_type,time_created FROM event_occurrence ORDER BY time_created,id")
  const eventFires = rows("SELECT id,definition_id,queue_position,event_job_revision_id,event_occurrence_id,causation_fire_id,causal_cycle,created_session_id,mission_opened_event_id,mission_disposition,mission_closure_event_id,time_created FROM event_job_fire ORDER BY definition_id,queue_position,id")
  const eventFireReceipts = rows("SELECT id,fire_id,definition_id,queue_position,outcome,disposition,closure_event_id,message_id,retry_at,time_created FROM event_job_fire_receipt ORDER BY time_created,id")
  return { owners,messages,events,inbox,deliveryReceipts,unresolvedInbox,leases,leaseGrants,tasks,taskIngress,taskIngressPolicies,taskWaits,taskWaitSettlements,sessionControls,sessionControlEvents,missionClosures,automationDefinitions,automationTombstones,automationTargets,automationFires,automationFrontiers,automationAttempts,automationAttemptReceipts,automationRuns,automationRunReceipts,automationDelaySettlements,eventDefinitions,eventTombstones,eventOccurrences,eventFires,eventFireReceipts }
})()
db.close()
const evidence = { observedAt: new Date().toISOString(),phase,database,readonly:true,
  boundary:"One coherent owned-database read of canonical durable IDs/occurrences/receipts/leases. No payload, credentials or tool keys. Unresolved inbox and persisted frontier are facts, not proof of active scheduling; no process-local registry imported or inferred from empty maps.",...facts }
fs.writeFileSync(path.join(evidenceRoot,`${prefix}-durable-${phase}.json`),JSON.stringify(evidence,null,2),{flag:"wx"})
console.log(JSON.stringify({phase,database,evidence:path.join(evidenceRoot,`${prefix}-durable-${phase}.json`),readonly:true}))
