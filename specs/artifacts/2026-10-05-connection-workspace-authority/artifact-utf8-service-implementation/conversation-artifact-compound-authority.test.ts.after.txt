import { afterEach, expect, test } from "bun:test"
import { createHash } from "node:crypto"
import {
  ApiAuthorityChangedError,
  ApiError,
  captureApiAuthority,
  configure,
  renewApiAuthority,
} from "../src/services/api"
import { loadConversationArtifactContent } from "../src/services/conversation-artifact"
import { __setHostTransportForTest } from "../src/services/host-transport-runtime"
import type { HostTransport, TransportRequest, TransportResponse } from "../src/services/host-transport"
import { createTauriTransport } from "../src/services/tauri-transport"
import { setBoardStore } from "../src/store/board"
import { applySettings, DEFAULT_SETTINGS, setSettingsStore } from "../src/store/settings"

const locator = { source: "task_artifact_resource", ref: { path: "authority.txt" } }
const encoder = new TextEncoder()
function setup() {
  configure({ serverUrl: "http://artifact-a.invalid", username: "dummy", password: "", directory: "D:/artifact-a" })
  setSettingsStore("directory", "D:/artifact-a")
  setBoardStore({ selectedSource: null, board: null })
  return captureApiAuthority()
}
function transport(request: (input: TransportRequest) => Promise<TransportResponse<Uint8Array>>) {
  __setHostTransportForTest({ ...createTauriTransport("browser"), request } as HostTransport)
}
function response(
  text: string,
  start: number,
  total: number,
  digest: string,
  attachment = false,
): TransportResponse<Uint8Array> {
  const bytes = encoder.encode(text)
  return {
    ok: true,
    status: 200,
    headers: {
      "content-type": attachment ? "application/octet-stream" : "text/plain",
      "content-disposition": attachment ? "attachment; filename*=UTF-8''authority.bin" : "inline",
      etag: `"sha256:${digest}"`,
      "content-range": total === 0 ? "bytes */0" : `bytes ${start}-${start + bytes.byteLength - 1}/${total}`,
    },
    body: bytes,
  }
}
afterEach(() => {
  __setHostTransportForTest(undefined)
  applySettings({ ...DEFAULT_SETTINGS })
  configure({ serverUrl: "http://127.0.0.1:7878", username: "opencorvus", password: "", directory: "" })
  setBoardStore({ selectedSource: null, board: null })
})

test("one artifact read retains entry directory, authority and exact complete data across chunks", async () => {
  const authority = setup()
  const value = "first\nsecond"
  const digest = createHash("sha256").update(value).digest("hex")
  const requests: TransportRequest[] = []
  transport(async (request) => {
    requests.push(request)
    setSettingsStore("directory", "D:/artifact-other")
    const first = requests.length === 1
    return response(first ? "first\n" : "second", first ? 0 : 6, 12, digest)
  })
  expect(await loadConversationArtifactContent({ taskID: "tsk_artifact", locator })).toEqual({
    locator,
    mediaType: "text/plain",
    sha256: digest,
    totalBytes: 12,
    text: value,
  })
  expect(
    requests.map((request) => ({ authority: request.authority, query: request.query, body: request.body })),
  ).toEqual(
    [0, 6].map((offset) => ({
      authority,
      query: { directory: "D:/artifact-a" },
      body: { kind: "json", value: { locator, byte_offset: offset, max_bytes: 65536 } },
    })),
  )
})

test.each(["renewal", "ABA"])("retains entry authority when %s occurs between accepted chunks", async (rotation) => {
  const authority = setup()
  const requests: TransportRequest[] = []
  transport(async (request) => {
    requests.push(request)
    const result = response("first\n", 0, 12, "a".repeat(64))
    const bytes = result.body
    // apiRequest has already checked this response when the reader consumes its body.
    Object.defineProperty(result, "body", {
      get() {
        if (rotation === "ABA") {
          configure({ serverUrl: "http://artifact-b.invalid" })
          configure({ serverUrl: "http://artifact-a.invalid" })
        } else renewApiAuthority()
        return bytes
      },
    })
    return result
  })
  const error: unknown = await loadConversationArtifactContent({ taskID: "tsk_artifact", locator }).catch(
    (error: unknown) => error,
  )
  expect(error).toBeInstanceOf(ApiAuthorityChangedError)
  const retired = error as ApiAuthorityChangedError
  expect({ phase: retired.phase, expected: retired.expectedRevision, current: retired.currentRevision }).toEqual({
    phase: "before_dispatch",
    expected: authority.revision,
    current: authority.revision + (rotation === "ABA" ? 2 : 1),
  })
  expect(
    requests.map((request) => ({ path: request.path, authority: request.authority, query: request.query })),
  ).toEqual([{ path: "task/tsk_artifact/artifact-read", authority, query: { directory: "D:/artifact-a" } }])
})

test("an admitted late chunk preserves its original response in typed retirement", async () => {
  const authority = setup()
  const entered = Promise.withResolvers<void>()
  const held = Promise.withResolvers<TransportResponse<Uint8Array>>()
  const requests: TransportRequest[] = []
  transport(async (request) => {
    requests.push(request)
    entered.resolve()
    return held.promise
  })
  const result = loadConversationArtifactContent({ taskID: "tsk_artifact", locator }).catch((error: unknown) => error)
  await entered.promise
  renewApiAuthority()
  const original = response("ok", 0, 2, "b".repeat(64))
  held.resolve(original)
  const error = (await result) as ApiAuthorityChangedError
  expect(error).toBeInstanceOf(ApiAuthorityChangedError)
  expect(error.outcome).toEqual({ phase: "response", response: original })
  expect(requests.map((request) => request.authority)).toEqual([authority])
})

test.each([false, true])("final %s attachment decision checks the original read authority", async (attachment) => {
  const authority = setup()
  transport(async () => {
    const result = response("ok", 0, 2, "c".repeat(64), attachment)
    const bytes = result.body
    Object.defineProperty(result, "body", {
      get() {
        renewApiAuthority()
        return bytes
      },
    })
    return result
  })
  const error = (await loadConversationArtifactContent({ taskID: "tsk_artifact", locator }).catch(
    (error: unknown) => error,
  )) as ApiAuthorityChangedError
  expect(error).toBeInstanceOf(ApiAuthorityChangedError)
  expect({ phase: error.phase, expected: error.expectedRevision, current: error.currentRevision }).toEqual({
    phase: "before_dispatch",
    expected: authority.revision,
    current: authority.revision + 1,
  })
})

test("current artifact byte rejection retains actual status and decoded body", async () => {
  setup()
  const body = encoder.encode("current rejected locator")
  transport(async () => ({ ok: false, status: 422, headers: { "content-type": "text/plain" }, body }))
  const error = (await loadConversationArtifactContent({ taskID: "tsk_artifact", locator }).catch(
    (error: unknown) => error,
  )) as ApiError
  expect(error).toBeInstanceOf(ApiError)
  expect(error.status).toBe(422)
  expect(error.message).toContain("current rejected locator")
})

test("current attachment and empty text preserve their exact return contracts", async () => {
  const authority = setup()
  const requests: TransportRequest[] = []
  transport(async (request) => {
    requests.push(request)
    return response("ok", 0, 2, "d".repeat(64), true)
  })
  expect(await loadConversationArtifactContent({ taskID: "tsk_artifact", locator })).toEqual({
    locator,
    mediaType: "application/octet-stream",
    sha256: "d".repeat(64),
    totalBytes: 2,
    bytes: encoder.encode("ok"),
    filename: "authority.bin",
  })
  transport(async (request) => {
    requests.push(request)
    return response("", 0, 0, "e".repeat(64))
  })
  expect(await loadConversationArtifactContent({ taskID: "tsk_artifact", locator })).toEqual({
    locator,
    mediaType: "text/plain",
    sha256: "e".repeat(64),
    totalBytes: 0,
    text: "",
  })
  expect(requests.map((request) => ({ authority: request.authority, query: request.query }))).toEqual([
    { authority, query: { directory: "D:/artifact-a" } },
    { authority, query: { directory: "D:/artifact-a" } },
  ])
})

test.each(["route boundary", "split bytes", "literal BOM"])("exact Unicode bytes survive %s chunks", async (kind) => {
  setup()
  const value = kind === "literal BOM" ? "\uFEFF" + "a".repeat(65533) + "\uFEFFtail" : "a".repeat(65535) + "中😀tail"
  const bytes = encoder.encode(value)
  const boundary = kind === "route boundary" ? 65535 : 65536
  const offsets: number[] = []
  transport(async (request) => {
    const body = request.body
    if (body?.kind !== "json") throw new Error("Expected actual JSON range request")
    const start = (body.value as { byte_offset: number }).byte_offset
    offsets.push(start)
    const end = start === 0 ? boundary : bytes.length
    return {
      ok: true,
      status: 200,
      headers: {
        "content-type": "text/plain",
        "content-disposition": "inline",
        etag: `"sha256:${"f".repeat(64)}"`,
        "content-range": `bytes ${start}-${end - 1}/${bytes.length}`,
      },
      body: bytes.slice(start, end),
    }
  })
  const result = await loadConversationArtifactContent({ taskID: "tsk_artifact", locator })
  expect(result).toEqual({
    locator,
    mediaType: "text/plain",
    sha256: "f".repeat(64),
    totalBytes: bytes.length,
    text: value,
  })
  expect(encoder.encode(result.text!)).toEqual(bytes)
  expect(offsets).toEqual([0, boundary])
})

test.each([
  [0xc3, 0x28],
  [0xf0, 0x9f],
])("invalid final UTF-8 bytes %j yield native TypeError", async (first, second) => {
  setup()
  const bytes = new Uint8Array([first, second])
  transport(async () => ({
    ok: true,
    status: 200,
    headers: {
      "content-type": "text/plain",
      "content-disposition": "inline",
      etag: `"sha256:${"f".repeat(64)}"`,
      "content-range": `bytes 0-${bytes.length - 1}/${bytes.length}`,
    },
    body: bytes,
  }))
  const error: unknown = await loadConversationArtifactContent({ taskID: "tsk_artifact", locator }).catch(
    (error: unknown) => error,
  )
  expect(error).toBeInstanceOf(TypeError)
})

test("supplied original authority validates entry and current completion", async () => {
  const authority = setup()
  transport(async () => response("ok", 0, 2, "d".repeat(64)))
  expect(await loadConversationArtifactContent({ taskID: "tsk_artifact", locator, authority })).toMatchObject({
    text: "ok",
    totalBytes: 2,
  })
  renewApiAuthority()
  const error = (await loadConversationArtifactContent({ taskID: "tsk_artifact", locator, authority }).catch(
    (error: unknown) => error,
  )) as ApiAuthorityChangedError
  expect(error).toBeInstanceOf(ApiAuthorityChangedError)
  expect({ phase: error.phase, expected: error.expectedRevision, current: error.currentRevision }).toEqual({
    phase: "before_dispatch",
    expected: authority.revision,
    current: authority.revision + 1,
  })
})

test("wrong actual range yields precise current continuity error", async () => {
  setup()
  transport(async () => response("ok", 1, 3, "d".repeat(64)))
  const error = (await loadConversationArtifactContent({ taskID: "tsk_artifact", locator }).catch(
    (error: unknown) => error,
  )) as Error
  expect({ name: error.name, message: error.message }).toEqual({
    name: "Error",
    message: "Artifact read returned a discontinuous byte range",
  })
})
