+export type BenchmarkInput = { profile: "fixture" | "ordinary"; output: string; runRoot?: string; port: number }
+export function benchmarkInput(args: readonly string[]): BenchmarkInput {
+  const allowed = new Set(["--profile", "--output", "--run-root", "--port"])
+  for (let index = 0; index < args.length; index += 2) {
+    assert(allowed.has(args[index]), `Unknown benchmark argument ${args[index]}`)
+    assert(args[index + 1] && !args[index + 1].startsWith("--"), `Missing value for ${args[index]}`)
+    assert.equal(args.indexOf(args[index]), index, `Duplicate benchmark argument ${args[index]}`)
+  }
+  const value = (name: string) => {
+    const index = args.indexOf(name)
+    return index < 0 ? undefined : args[index + 1]
+  }
+  const profile = value("--profile")
+  assert(profile === "fixture" || profile === "ordinary", "--profile fixture|ordinary is required")
+  const output = value("--output")
+  assert(output, "--output <fresh-evidence-directory> is required")
+  const runRoot = value("--run-root")
+  if (profile === "ordinary") assert(runRoot, "ordinary requires --run-root <fresh-owned-root>")
+  const port = Number(value("--port") ?? "17947")
+  assert(Number.isInteger(port) && port > 0 && port <= 65535, "Valid loopback port is required")
+  return { profile, output: path.resolve(output), ...(runRoot ? { runRoot: path.resolve(runRoot) } : {}), port }

+export function observedRequestID(rows: readonly Record<string, any>[], responseID?: string | null): string {
