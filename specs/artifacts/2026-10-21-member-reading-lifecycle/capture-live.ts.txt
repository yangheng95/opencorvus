import fs from 'node:fs'
import {Database} from 'bun:sqlite'

 })
 const readers=audit.requests.map((item:any)=>({model:item.model,status:item.status,streaming:item.streaming,reader:item.response_reader})),frame=process.argv[2]??'status'

 fs.writeFileSync(root+'/frame-'+frame+'.json',JSON.stringify(result,null,2)+'\n')
 console.log(JSON.stringify({observedAt:result.observedAt,requests:result.requests,maxRequests:result.maximumScopeRequests,messages,readers:readers.map(({model,status,streaming,reader}:any)=>({model,status,streaming,state:reader.state,terminal:reader.terminal?.kind,sessionID:reader.requestContext?.sessionID,agent:reader.requestContext?.streamRequest?.agentID,assistantID:reader.requestContext?.activity?.assistantMessageID}))}))
