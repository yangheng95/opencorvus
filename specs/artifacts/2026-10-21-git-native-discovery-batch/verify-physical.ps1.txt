}
@{observedAtUtc=[DateTime]::UtcNow.ToString('o');scopes=$taskRecords;count=$taskRecords.Count;boundary='Eight original native/output/request/launcher settlements;no restarted occurrence'}|ConvertTo-Json -Depth 15|Set-Content -LiteralPath "$taskRoot/physical-scopes.json"
